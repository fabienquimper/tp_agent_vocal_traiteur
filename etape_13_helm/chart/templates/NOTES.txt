     make k8s-init-ollama

     # Ou directement dans le pod :

     make k8s-configmaps

  4. Tester l'API :
     curl http://localhost:8000/health

       -H "Content-Type: application/json" \
       -d '{"text": "Bonjour, que proposez-vous ?"}'

  COMMANDES HELM UTILES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  # Voir les valeurs actives
  helm get values {{ .Release.Name }}

  # Upgrade sans interruption (rolling update automatique)
  helm upgrade {{ .Release.Name }} ./chart

  # Upgrade avec surcharge : passer en staging
  helm upgrade {{ .Release.Name }} ./chart -f chart/values.staging.yaml

  # Activer le canary (10% du trafic)
  helm upgrade {{ .Release.Name }} ./chart --set canary.enabled=true

  # Basculer STT vers le modèle Whisper small (green)
  helm upgrade {{ .Release.Name }} ./chart --set stt.activeColor=green

  # Rollback à la version précédente
  helm rollback {{ .Release.Name }}

  # Historique des releases
  helm history {{ .Release.Name }}

  # Désinstaller (supprime TOUTES les ressources K8s du chart)
  helm uninstall {{ .Release.Name }} -n {{ .Values.namespace }}
