# ─── OpenTelemetry — Traces distribuées (étape 10) ───────────────────────────
#
# Pourquoi OTel et pas Prometheus seul ?
#   Prometheus = métriques agrégées (counters, histograms sur du temps)
#   OTel traces = voyage d'UNE requête à travers les services (causalité)
#
# Stack :
#   opentelemetry-sdk            → TracerProvider, Span, BatchSpanProcessor
#   opentelemetry-exporter-otlp  → envoi des spans vers Jaeger via OTLP/gRPC
#   opentelemetry-instrumentation-fastapi → auto-instrumentation HTTP entrant
#   opentelemetry-instrumentation-httpx   → auto-instrumentation HTTP sortant
#
# Les versions instrumentations suivent le pattern 0.{sdk_minor}b0
# Pour SDK 1.24.0 → instrumentations 0.45b0
opentelemetry-api==1.24.0
opentelemetry-sdk==1.24.0
opentelemetry-exporter-otlp-proto-grpc==1.24.0
opentelemetry-instrumentation-fastapi==0.45b0
opentelemetry-instrumentation-httpx==0.45b0
